cmake_minimum_required(VERSION 3.18)
project(PM LANGUAGES CXX )
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include)
find_package(OpenMP REQUIRED)
option(PM_ENABLE_CUDA "Build CUDA benchmarks when a CUDA compiler is available" ON)

add_subdirectory(test)

# find_package(MPI REQUIRED)
# add_subdirectory(test_mpi)

if(PM_ENABLE_CUDA)
    include(CheckLanguage)
    check_language(CUDA)
    if(CMAKE_CUDA_COMPILER)
        enable_language(CUDA)
        set(CMAKE_CUDA_ARCHITECTURES native)
        add_subdirectory(test_cuda)
        message(STATUS "CUDA compiler detected, enabling test_cuda")
    else()
        message(STATUS "CUDA compiler not found, building CPU-only targets")
    endif()
else()
    message(STATUS "PM_ENABLE_CUDA=OFF, building CPU-only targets")
endif()

install(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/include/ DESTINATION include)
